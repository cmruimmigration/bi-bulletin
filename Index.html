        .status-container { display: flex; justify-content: space-between; align-items: center; color: #888; font-size: 0.9rem; }

        .loading { font-style: italic; }

    </style>

<body>


    <h1>Live Data Dashboard</h1>

    

    <div class="status-container">

        <span id="loading-text" class="loading">Connecting to spreadsheet...</span>

        <span id="last-updated"></span>

    </div>

    

    <table id="data-table" style="display: none;">

        <thead id="table-head"></thead>

        <tbody id="table-body"></tbody>

    </table>


    <script>

        // 1. PASTE YOUR GOOGLE SHEET CSV LINK BETWEEN THE QUOTES BELOW:

        const baseCsvUrl = "YOUR_PUBLISHED_CSV_URL_HERE";


        async function fetchSheetData() {

            try {

                // Append a unique timestamp (?t=...) to force Google to give us the absolute freshest data

                const freshUrl = `${baseCsvUrl}&t=${new Date().getTime()}`;

                

                const response = await fetch(freshUrl);

                const csvText = await response.text();

                

                // Parse CSV text into rows and elements cleanly

                const rows = csvText.split("\n").map(row => {

                    return row.split(/,(?=(?:(?:[^"]*"){2})*[^"]*\()/).map(cell => cell.replace(/^"|"\$/g, '').trim());

                });


                if (rows.length === 0 || (rows.length === 1 && rows[0][0] === "")) {

                    throw new Error("Spreadsheet appears to be empty.");

                }


                const tableHead = document.getElementById("table-head");

                const tableBody = document.getElementById("table-body");


                // 2. Build Table Headers (Row 1 of your sheet)

                const headers = rows[0];

                let headHtml = "<tr>";

                headers.forEach(header => {

                    headHtml += `<th>${header}</th>`;

                });

                headHtml += "</tr>";

                tableHead.innerHTML = headHtml;


                // 3. Build Data Rows (Rows 2 and beyond)

                let bodyHtml = "";

                for (let i = 1; i < rows.length; i++) {

                    if (rows[i].length === 1 && rows[i][0] === "") continue; // Skip blank rows

                    

                    bodyHtml += "<tr>";

                    rows[i].forEach(cell => {

                        bodyHtml += `<td>${cell || ""}</td>`;

                    });

                    bodyHtml += "</tr>";

                }

                tableBody.innerHTML = bodyHtml;


                // Update status text and display the completed table

                document.getElementById("loading-text").innerText = "● Live connection active";

                document.getElementById("loading-text").style.color = "#4caf50"; // Green status light

                document.getElementById("last-updated").innerText = `Last update: ${new Date().toLocaleTimeString()}`;

                document.getElementById("data-table").style.display = "table";


            } catch (error) {

                console.error("Error fetching data:", error);

                document.getElementById("loading-text").innerText = "⚠️ Failed to sync. Checking link connection...";

                document.getElementById("loading-text").style.color = "#f44336"; // Red status light

            }

        }


        // Run immediately when someone opens the page

        fetchSheetData();


        // Auto-refresh the data every 30 seconds seamlessly without blinking the page

        setInterval(fetchSheetData, 30000);

    </script>

</body>
